WITH previous_snapshots AS (

    SELECT *
    FROM TABLE(
        previous.query.result(
            schema => DESCRIPTOR(
                snapshot_date DATE,
                magic_eden_floor_sol DOUBLE,
                aggregated_listed_count BIGINT
            )
        )
    )

),

tensor_history AS (

    SELECT
        CAST(snapshot_date AS DATE) AS snapshot_date,
        CAST(floor_close_sol AS DOUBLE) AS magic_eden_floor_sol,
        CAST(listed_count AS BIGINT) AS aggregated_listed_count

    FROM dune.cjohnson.dataset_guild_saga_tensor_history

),

combined_history AS (

    -- Keep everything already accumulated by this scheduled query.
    SELECT
        snapshot_date,
        magic_eden_floor_sol,
        aggregated_listed_count
    FROM previous_snapshots

    UNION ALL

    -- Backfill Tensor history only for dates earlier than our
    -- earliest already-collected daily snapshot.
    SELECT
        t.snapshot_date,
        t.magic_eden_floor_sol,
        t.aggregated_listed_count
    FROM tensor_history t
    WHERE t.snapshot_date < COALESCE(
        (
            SELECT MIN(snapshot_date)
            FROM previous_snapshots
        ),
        DATE '9999-12-31'
    )

),

magic_eden_only AS (

    SELECT http_get(
        'https://api-mainnet.magiceden.dev/v2/collections/guild_saga_heroes/stats?listingAggMode=false'
    ) AS response

),

aggregated AS (

    SELECT http_get(
        'https://api-mainnet.magiceden.dev/v2/collections/guild_saga_heroes/stats?listingAggMode=true'
    ) AS response

),

today_snapshot AS (

    SELECT
        CURRENT_DATE AS snapshot_date,

        CAST(
            json_extract_scalar(
                json_parse(me.response),
                '$.floorPrice'
            ) AS DOUBLE
        ) / 1000000000 AS magic_eden_floor_sol,

        CAST(
            json_extract_scalar(
                json_parse(agg.response),
                '$.listedCount'
            ) AS BIGINT
        ) AS aggregated_listed_count

    FROM magic_eden_only me
    CROSS JOIN aggregated agg

)

-- Preserve all historical dates except today's old value.
SELECT
    snapshot_date,
    magic_eden_floor_sol,
    aggregated_listed_count
FROM combined_history
WHERE snapshot_date < CURRENT_DATE

UNION ALL

-- Replace today's value with the current live Magic Eden snapshot.
SELECT
    snapshot_date,
    magic_eden_floor_sol,
    aggregated_listed_count
FROM today_snapshot

ORDER BY snapshot_date DESC
