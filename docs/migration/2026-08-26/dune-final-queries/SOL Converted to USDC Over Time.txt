WITH swaps AS (
    SELECT *
    FROM (
        VALUES
            (TIMESTAMP '2022-03-02 03:56:01', 770.000010, 79328.404560),
            (TIMESTAMP '2022-03-19 17:51:25', 550.000010, 50675.638785),
            (TIMESTAMP '2022-03-24 23:34:35', 500.000010, 50378.622377),
            (TIMESTAMP '2022-03-28 20:44:17', 450.000010, 49570.498336),
            (TIMESTAMP '2022-03-31 04:54:47', 420.000010, 50287.715199),

            (TIMESTAMP '2022-04-04 04:57:59', 375.000010, 50318.357107),

            (TIMESTAMP '2022-05-15 23:25:26', 850.000010, 48775.104645),

            (TIMESTAMP '2022-06-16 03:55:51', 1200.000010, 40004.679168),

            (TIMESTAMP '2023-09-04 20:51:59', 150.000016, 2911.380059),
            (TIMESTAMP '2023-09-18 04:31:08', 400.000018, 7619.904121),

            (TIMESTAMP '2023-10-02 16:31:35', 400.000016, 9633.380238),
            (TIMESTAMP '2023-10-18 01:07:27', 400.000016, 9586.328282),

            (TIMESTAMP '2023-11-11 08:01:38', 450.000015, 24466.133035),

            (TIMESTAMP '2025-03-02 18:04:58', 85.000704, 14378.595419),
            (TIMESTAMP '2025-03-07 00:28:52', 83.901915, 11548.837117)
    ) AS t(swap_utc, sol_sold, usdc_purchased)
),

monthly AS (
    SELECT
        CAST(DATE_TRUNC('month', swap_utc) AS DATE) AS month,
        SUM(sol_sold) AS sol_sold,
        SUM(usdc_purchased) AS usdc_purchased
    FROM swaps
    GROUP BY 1
),

calendar AS (
    SELECT month
    FROM UNNEST(
        SEQUENCE(
            DATE '2022-03-01',
            DATE '2025-03-01',
            INTERVAL '1' MONTH
        )
    ) AS t(month)
),

filled AS (
    SELECT
        c.month,
        COALESCE(m.sol_sold, 0) AS sol_sold,
        COALESCE(m.usdc_purchased, 0) AS usdc_purchased
    FROM calendar c
    LEFT JOIN monthly m
        ON c.month = m.month
)

SELECT
    month,

    ROUND(sol_sold, 3) AS sol_sold,

    ROUND(
        SUM(sol_sold) OVER (
            ORDER BY month
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        ),
        3
    ) AS total_sol_sold,

    ROUND(usdc_purchased, 2) AS usdc_purchased,

    ROUND(
        SUM(usdc_purchased) OVER (
            ORDER BY month
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        ),
        2
    ) AS total_usdc_purchased

FROM filled
ORDER BY month;
