WITH prev AS (

    SELECT *
    FROM TABLE(
        previous.query.result(
            schema => DESCRIPTOR(
                signature VARCHAR,
                mint VARCHAR,
                hero_number INTEGER,
                hero_name VARCHAR,

                block_time BIGINT,
                utc VARCHAR,
                sale_date VARCHAR,
                sale_month VARCHAR,
                sale_year INTEGER,

                marketplace VARCHAR,
                marketplace_detail VARCHAR,
                marketplace_attribution_method VARCHAR,
                direct_program_families VARCHAR,

                buyer VARCHAR,
                seller VARCHAR,

                gross_price_sol DOUBLE,
                gross_price_method VARCHAR,

                price_recovered_in_v3 INTEGER,
                sale_class VARCHAR,
                legacy_sale INTEGER,

                royalty_90_sol DOUBLE,
                royalty_10_sol DOUBLE,
                royalty_total_sol DOUBLE,
                royalty_paid INTEGER,

                collector_checkpoint_date DATE,

                kpi_secondary_sales BIGINT,
                kpi_secondary_volume_sol DOUBLE,
                kpi_unique_buyers BIGINT,
                kpi_heroes_ever_sold BIGINT
            )
        )
    )
),

/* ============================================================
   CHECKPOINT

   Two-day overlap:
   previous checkpoint date + previous calendar day through today.

   If a scheduled run is missed, the checkpoint stays old,
   so the next successful execution automatically covers
   the missing dates.
   ============================================================ */

checkpoint AS (

    SELECT
        COALESCE(
            MAX(collector_checkpoint_date),
            DATE '2026-08-19'
        ) AS last_checkpoint_date,

        date_add(
            'day',
            -1,
            COALESCE(
                MAX(collector_checkpoint_date),
                DATE '2026-08-19'
            )
        ) AS scan_start_date

    FROM prev
),

/* ============================================================
   TRUSTED HISTORICAL BASELINE
   ============================================================ */

baseline AS (

    SELECT
        CAST(signature AS VARCHAR)
            AS signature,

        CAST(mint AS VARCHAR)
            AS mint,

        CAST(hero_number AS INTEGER)
            AS hero_number,

        CAST(hero_name AS VARCHAR)
            AS hero_name,

        CAST(block_time AS BIGINT)
            AS block_time,

        CAST(utc AS VARCHAR)
            AS utc,

        CAST(sale_date AS VARCHAR)
            AS sale_date,

        CAST(sale_month AS VARCHAR)
            AS sale_month,

        CAST(sale_year AS INTEGER)
            AS sale_year,

        CAST(marketplace AS VARCHAR)
            AS marketplace,

        CAST(marketplace_detail AS VARCHAR)
            AS marketplace_detail,

        CAST(
            marketplace_attribution_method AS VARCHAR
        ) AS marketplace_attribution_method,

        CAST(
            direct_program_families AS VARCHAR
        ) AS direct_program_families,

        CAST(buyer AS VARCHAR)
            AS buyer,

        CAST(seller AS VARCHAR)
            AS seller,

        CAST(gross_price_sol AS DOUBLE)
            AS gross_price_sol,

        CAST(gross_price_method AS VARCHAR)
            AS gross_price_method,

        CAST(price_recovered_in_v3 AS INTEGER)
            AS price_recovered_in_v3,

        CAST(sale_class AS VARCHAR)
            AS sale_class,

        CAST(legacy_sale AS INTEGER)
            AS legacy_sale,

        CAST(royalty_90_sol AS DOUBLE)
            AS royalty_90_sol,

        TRY_CAST(royalty_10_sol AS DOUBLE)
            AS royalty_10_sol,

        CAST(royalty_total_sol AS DOUBLE)
            AS royalty_total_sol,

        CAST(royalty_paid AS INTEGER)
            AS royalty_paid,

        0 AS source_priority

    FROM dune.cjohnson.dataset_guild_saga_market_sales_final
),

/* ============================================================
   PREVIOUS LIVE SALES ONLY

   Historical baseline rows do not need to feed themselves
   through previous.query.result() forever.

   Only post-baseline live sales need persistence.

   Priority:
       0 = historical baseline
       1 = freshly detected live sale
       2 = previously preserved live sale
   ============================================================ */

previous_live_sales AS (

    SELECT
        signature,
        mint,
        hero_number,
        hero_name,

        block_time,
        utc,
        sale_date,
        sale_month,
        sale_year,

        marketplace,
        marketplace_detail,
        marketplace_attribution_method,
        direct_program_families,

        buyer,
        seller,

        gross_price_sol,
        gross_price_method,

        price_recovered_in_v3,
        sale_class,
        legacy_sale,

        royalty_90_sol,
        royalty_10_sol,
        royalty_total_sol,
        royalty_paid,

        2 AS source_priority

    FROM prev

    WHERE sale_class = 'LIVE_DETECTED_SALE'
),

/* ============================================================
   RECENT MARKETPLACE INSTRUCTIONS

   Three currently-active Guild Saga marketplace paths:

       Magic Eden V2
       Tensor Marketplace
       Tensor AMM

   Only the validated buy/sale + price-event discriminators
   are retained.
   ============================================================ */

recent_market_calls AS (

    SELECT
        i.block_date,
        i.block_time,
        i.tx_id,
        i.executing_account,

        i.outer_instruction_index,
        i.account_arguments,
        i.data,

        varbinary_substring(
            i.data,
            1,
            8
        ) AS discriminator

    FROM solana.instruction_calls AS i

    WHERE
        i.block_date >= (
            SELECT scan_start_date
            FROM checkpoint
        )

        AND i.block_date <= CURRENT_DATE

        AND i.tx_success = TRUE

        AND i.executing_account IN (
            'M2mx93ekt1fmXSVkTrUL9xVFHkmME8HTUi5Cyc5aF7K',
            'TCMPhJdwDryooaGtiocG1u3xcYbRpiJzb283XfCZsDp',
            'TAMM6ub33ij1mbetoMyVBLeKY5iP41i4UPUJQGkhfsg'
        )

        AND varbinary_substring(
            i.data,
            1,
            8
        ) IN (
            /* Magic Eden ExecuteSaleV2 */
            0x5bdc31dfcc8135c1,

            /* Tensor Marketplace BuyLegacy */
            0x447f2b08d41ff972,

            /* Tensor Marketplace TakeEvent */
            0x6aa20ae28444df15,

            /* Tensor AMM BuyNft */
            0x60001cbe316b53de,

            /* Tensor AMM BuySellEvent */
            0x1fa2e49e99a0c6b6
        )
),

/* ============================================================
   RECENT ROYALTY TRANSFERS

   Validated System Program transfer layout:

       instruction type = 2
       recipient        = account argument 2
       lamports         = data bytes 5-12

   Guild Saga receives the 90% creator-royalty share.
   RRUM receives the remaining 10% creator-royalty share.
   ============================================================ */

recent_royalty_transfers AS (

    SELECT
        i.tx_id,

        SUM(
            CASE
                WHEN element_at(
                    i.account_arguments,
                    2
                ) =
                '8VAHrpJ9nsqRLaujbzpuCxhAhjsE8wA4ZvUHwx2VZw3y'

                THEN
                    from_big_endian_64(
                        reverse(
                            varbinary_substring(
                                i.data,
                                5,
                                8
                            )
                        )
                    )

                ELSE BIGINT '0'
            END
        ) AS royalty_90_lamports,

        SUM(
            CASE
                WHEN element_at(
                    i.account_arguments,
                    2
                ) =
                'RRUMF9KYPcvNSmnicNMAFKx5wDYix3wjNa6bA7R6xqA'

                THEN
                    from_big_endian_64(
                        reverse(
                            varbinary_substring(
                                i.data,
                                5,
                                8
                            )
                        )
                    )

                ELSE BIGINT '0'
            END
        ) AS royalty_10_lamports

    FROM solana.instruction_calls AS i

    WHERE
        i.block_date >= (
            SELECT scan_start_date
            FROM checkpoint
        )

        AND i.block_date <= CURRENT_DATE

        AND i.tx_success = TRUE

        AND i.executing_account =
            '11111111111111111111111111111111'

        AND element_at(
            i.account_arguments,
            2
        ) IN (
            '8VAHrpJ9nsqRLaujbzpuCxhAhjsE8wA4ZvUHwx2VZw3y',
            'RRUMF9KYPcvNSmnicNMAFKx5wDYix3wjNa6bA7R6xqA'
        )

        AND from_big_endian_32(
            reverse(
                varbinary_substring(
                    i.data,
                    1,
                    4
                )
            )
        ) = 2

    GROUP BY
        i.tx_id
),

/* ============================================================
   GROUP MARKETPLACE BUY + PRICE EVENT INTO ONE SALE

   Tensor buy instructions and their corresponding price events
   share the same tx_id + outer_instruction_index.

   Magic Eden ExecuteSaleV2 contains everything required
   directly in the sale instruction.
   ============================================================ */

decoded_groups AS (

    SELECT
        block_time,
        tx_id,
        outer_instruction_index,

        /* ====================================================
           MAGIC EDEN V2
           ==================================================== */

        MAX(
            CASE
                WHEN executing_account =
                    'M2mx93ekt1fmXSVkTrUL9xVFHkmME8HTUi5Cyc5aF7K'

                AND discriminator =
                    0x5bdc31dfcc8135c1

                THEN 1
                ELSE 0
            END
        ) AS me_sale,

        MAX(
            CASE
                WHEN executing_account =
                    'M2mx93ekt1fmXSVkTrUL9xVFHkmME8HTUi5Cyc5aF7K'

                AND discriminator =
                    0x5bdc31dfcc8135c1

                THEN element_at(
                    account_arguments,
                    5
                )
            END
        ) AS me_mint,

        MAX(
            CASE
                WHEN executing_account =
                    'M2mx93ekt1fmXSVkTrUL9xVFHkmME8HTUi5Cyc5aF7K'

                AND discriminator =
                    0x5bdc31dfcc8135c1

                THEN element_at(
                    account_arguments,
                    1
                )
            END
        ) AS me_buyer,

        MAX(
            CASE
                WHEN executing_account =
                    'M2mx93ekt1fmXSVkTrUL9xVFHkmME8HTUi5Cyc5aF7K'

                AND discriminator =
                    0x5bdc31dfcc8135c1

                THEN element_at(
                    account_arguments,
                    2
                )
            END
        ) AS me_seller,

        MAX(
            CASE
                WHEN executing_account =
                    'M2mx93ekt1fmXSVkTrUL9xVFHkmME8HTUi5Cyc5aF7K'

                AND discriminator =
                    0x5bdc31dfcc8135c1

                THEN
                    CAST(
                        from_big_endian_64(
                            reverse(
                                varbinary_substring(
                                    data,
                                    11,
                                    8
                                )
                            )
                        )
                        AS DOUBLE
                    )
                    / 1000000000.0
            END
        ) AS me_price_sol,

        /* ====================================================
           TENSOR MARKETPLACE
           ==================================================== */

        MAX(
            CASE
                WHEN executing_account =
                    'TCMPhJdwDryooaGtiocG1u3xcYbRpiJzb283XfCZsDp'

                AND discriminator =
                    0x447f2b08d41ff972

                THEN 1
                ELSE 0
            END
        ) AS tm_buy,

        MAX(
            CASE
                WHEN executing_account =
                    'TCMPhJdwDryooaGtiocG1u3xcYbRpiJzb283XfCZsDp'

                AND discriminator =
                    0x6aa20ae28444df15

                THEN 1
                ELSE 0
            END
        ) AS tm_event,

        MAX(
            CASE
                WHEN executing_account =
                    'TCMPhJdwDryooaGtiocG1u3xcYbRpiJzb283XfCZsDp'

                AND discriminator =
                    0x447f2b08d41ff972

                THEN element_at(
                    account_arguments,
                    6
                )
            END
        ) AS tm_mint,

        MAX(
            CASE
                WHEN executing_account =
                    'TCMPhJdwDryooaGtiocG1u3xcYbRpiJzb283XfCZsDp'

                AND discriminator =
                    0x447f2b08d41ff972

                THEN element_at(
                    account_arguments,
                    2
                )
            END
        ) AS tm_buyer,

        MAX(
            CASE
                WHEN executing_account =
                    'TCMPhJdwDryooaGtiocG1u3xcYbRpiJzb283XfCZsDp'

                AND discriminator =
                    0x447f2b08d41ff972

                THEN element_at(
                    account_arguments,
                    7
                )
            END
        ) AS tm_seller,

        MAX(
            CASE
                WHEN executing_account =
                    'TCMPhJdwDryooaGtiocG1u3xcYbRpiJzb283XfCZsDp'

                AND discriminator =
                    0x6aa20ae28444df15

                THEN
                    CAST(
                        from_big_endian_64(
                            reverse(
                                varbinary_substring(
                                    data,
                                    78,
                                    8
                                )
                            )
                        )
                        AS DOUBLE
                    )
                    / 1000000000.0
            END
        ) AS tm_price_sol,

        /* ====================================================
           TENSOR AMM
           ==================================================== */

        MAX(
            CASE
                WHEN executing_account =
                    'TAMM6ub33ij1mbetoMyVBLeKY5iP41i4UPUJQGkhfsg'

                AND discriminator =
                    0x60001cbe316b53de

                THEN 1
                ELSE 0
            END
        ) AS ta_buy,

        MAX(
            CASE
                WHEN executing_account =
                    'TAMM6ub33ij1mbetoMyVBLeKY5iP41i4UPUJQGkhfsg'

                AND discriminator =
                    0x1fa2e49e99a0c6b6

                THEN 1
                ELSE 0
            END
        ) AS ta_event,

        MAX(
            CASE
                WHEN executing_account =
                    'TAMM6ub33ij1mbetoMyVBLeKY5iP41i4UPUJQGkhfsg'

                AND discriminator =
                    0x60001cbe316b53de

                THEN element_at(
                    account_arguments,
                    15
                )
            END
        ) AS ta_mint,

        MAX(
            CASE
                WHEN executing_account =
                    'TAMM6ub33ij1mbetoMyVBLeKY5iP41i4UPUJQGkhfsg'

                AND discriminator =
                    0x60001cbe316b53de

                THEN element_at(
                    account_arguments,
                    2
                )
            END
        ) AS ta_buyer,

        MAX(
            CASE
                WHEN executing_account =
                    'TAMM6ub33ij1mbetoMyVBLeKY5iP41i4UPUJQGkhfsg'

                AND discriminator =
                    0x60001cbe316b53de

                THEN element_at(
                    account_arguments,
                    8
                )
            END
        ) AS ta_seller,

        MAX(
            CASE
                WHEN executing_account =
                    'TAMM6ub33ij1mbetoMyVBLeKY5iP41i4UPUJQGkhfsg'

                AND discriminator =
                    0x1fa2e49e99a0c6b6

                THEN
                    CAST(
                        from_big_endian_64(
                            reverse(
                                varbinary_substring(
                                    data,
                                    10,
                                    8
                                )
                            )
                        )
                        AS DOUBLE
                    )
                    / 1000000000.0
            END
        ) AS ta_price_sol

    FROM recent_market_calls

    GROUP BY
        block_time,
        tx_id,
        outer_instruction_index
),

/* ============================================================
   NORMALIZE THREE MARKETPLACE FAMILIES
   ============================================================ */

decoded_live AS (

    SELECT
        block_time,
        tx_id,

        CASE
            WHEN me_sale = 1
                THEN 'Magic Eden V2'

            WHEN
                tm_buy = 1
                AND tm_event = 1
                THEN 'Tensor Marketplace'

            WHEN
                ta_buy = 1
                AND ta_event = 1
                THEN 'Tensor AMM'
        END AS family,

        CASE
            WHEN me_sale = 1
                THEN me_mint

            WHEN
                tm_buy = 1
                AND tm_event = 1
                THEN tm_mint

            WHEN
                ta_buy = 1
                AND ta_event = 1
                THEN ta_mint
        END AS mint,

        CASE
            WHEN me_sale = 1
                THEN me_buyer

            WHEN
                tm_buy = 1
                AND tm_event = 1
                THEN tm_buyer

            WHEN
                ta_buy = 1
                AND ta_event = 1
                THEN ta_buyer
        END AS buyer,

        CASE
            WHEN me_sale = 1
                THEN me_seller

            WHEN
                tm_buy = 1
                AND tm_event = 1
                THEN tm_seller

            WHEN
                ta_buy = 1
                AND ta_event = 1
                THEN ta_seller
        END AS seller,

        CASE
            WHEN me_sale = 1
                THEN me_price_sol

            WHEN
                tm_buy = 1
                AND tm_event = 1
                THEN tm_price_sol

            WHEN
                ta_buy = 1
                AND ta_event = 1
                THEN ta_price_sol
        END AS gross_price_sol

    FROM decoded_groups

    WHERE
           me_sale = 1

        OR (
            tm_buy = 1
            AND tm_event = 1
        )

        OR (
            ta_buy = 1
            AND ta_event = 1
        )
),

/* ============================================================
   KEEP ONLY GUILD SAGA HERO SALES
   + ATTACH ROYALTY PAYMENTS
   ============================================================ */

live_sales AS (

    SELECT
        CAST(d.tx_id AS VARCHAR)
            AS signature,

        CAST(d.mint AS VARCHAR)
            AS mint,

        CAST(h.hero_number AS INTEGER)
            AS hero_number,

        CAST(h.hero_name AS VARCHAR)
            AS hero_name,

        CAST(
            to_unixtime(d.block_time)
            AS BIGINT
        ) AS block_time,

        CAST(d.block_time AS VARCHAR)
            AS utc,

        CAST(
            CAST(d.block_time AS DATE)
            AS VARCHAR
        ) AS sale_date,

        CAST(year(d.block_time) AS VARCHAR)
            || '-'
            || lpad(
                CAST(
                    month(d.block_time)
                    AS VARCHAR
                ),
                2,
                '0'
            )
            AS sale_month,

        CAST(
            year(d.block_time)
            AS INTEGER
        ) AS sale_year,

        CASE
            WHEN d.family = 'Magic Eden V2'
                THEN 'Magic Eden'

            ELSE 'Tensor'
        END AS marketplace,

        d.family
            AS marketplace_detail,

        'direct_program_live'
            AS marketplace_attribution_method,

        CASE
            WHEN d.family = 'Magic Eden V2'
                THEN 'Magic Eden v2'

            WHEN d.family = 'Tensor Marketplace'
                THEN 'Tensor Marketplace'

            WHEN d.family = 'Tensor AMM'
                THEN 'Tensor AMM'
        END AS direct_program_families,

        CAST(d.buyer AS VARCHAR)
            AS buyer,

        CAST(d.seller AS VARCHAR)
            AS seller,

        CAST(d.gross_price_sol AS DOUBLE)
            AS gross_price_sol,

        CASE
            WHEN d.family = 'Magic Eden V2'
                THEN 'magic_eden_execute_sale_v2_price'

            WHEN d.family = 'Tensor Marketplace'
                THEN 'tensor_tcomp_take_event_amount'

            WHEN d.family = 'Tensor AMM'
                THEN 'tensor_amm_buy_sell_event_current_price'
        END AS gross_price_method,

        0 AS price_recovered_in_v3,

        'LIVE_DETECTED_SALE'
            AS sale_class,

        0 AS legacy_sale,

        /* ====================================================
           ROYALTY — GUILD SAGA 90% SHARE
           ==================================================== */

        CAST(
            COALESCE(
                r.royalty_90_lamports,
                BIGINT '0'
            )
            AS DOUBLE
        ) / 1000000000.0
            AS royalty_90_sol,

        /* ====================================================
           ROYALTY — RRUM 10% SHARE
           ==================================================== */

        CAST(
            COALESCE(
                r.royalty_10_lamports,
                BIGINT '0'
            )
            AS DOUBLE
        ) / 1000000000.0
            AS royalty_10_sol,

        /* ====================================================
           TOTAL CREATOR ROYALTY
           ==================================================== */

        CAST(
            COALESCE(
                r.royalty_90_lamports,
                BIGINT '0'
            )
            +
            COALESCE(
                r.royalty_10_lamports,
                BIGINT '0'
            )
            AS DOUBLE
        ) / 1000000000.0
            AS royalty_total_sol,

        /* ====================================================
           DID THIS SALE ACTUALLY PAY A CREATOR ROYALTY?
           ==================================================== */

        CASE
            WHEN
                COALESCE(
                    r.royalty_90_lamports,
                    BIGINT '0'
                )
                +
                COALESCE(
                    r.royalty_10_lamports,
                    BIGINT '0'
                )
                > 0

            THEN 1
            ELSE 0
        END AS royalty_paid,

        1 AS source_priority

    FROM decoded_live AS d

    INNER JOIN
        dune.cjohnson.dataset_guild_saga_assets_v2 AS h
        ON h.mint = d.mint

    LEFT JOIN recent_royalty_transfers AS r
        ON r.tx_id = d.tx_id

    WHERE
        d.mint IS NOT NULL

        AND d.buyer IS NOT NULL

        AND d.seller IS NOT NULL

        AND d.gross_price_sol IS NOT NULL
),

/* ============================================================
   MERGE BASELINE + NEW LIVE SALES + PREVIOUS LIVE SALES
   ============================================================ */

all_candidates AS (

    SELECT *
    FROM baseline

    UNION ALL

    SELECT *
    FROM live_sales

    UNION ALL

    SELECT *
    FROM previous_live_sales
),

/* ============================================================
   DEDUPLICATE

   Historical baseline wins when an old historical sale happens
   to fall inside the live lookback.

   Fresh live detection wins over its previously-preserved copy.
   ============================================================ */

deduped AS (

    SELECT
        *,

        ROW_NUMBER() OVER (
            PARTITION BY
                signature,
                mint

            ORDER BY
                source_priority ASC
        ) AS duplicate_rank

    FROM all_candidates
),

final_sales AS (

    SELECT
        signature,
        mint,
        hero_number,
        hero_name,

        block_time,
        utc,
        sale_date,
        sale_month,
        sale_year,

        marketplace,
        marketplace_detail,
        marketplace_attribution_method,
        direct_program_families,

        buyer,
        seller,

        gross_price_sol,
        gross_price_method,

        price_recovered_in_v3,
        sale_class,
        legacy_sale,

        royalty_90_sol,
        royalty_10_sol,
        royalty_total_sol,
        royalty_paid

    FROM deduped

    WHERE duplicate_rank = 1
),

/* ============================================================
   DASHBOARD KPIs
   ============================================================ */

metrics AS (

    SELECT
        COUNT(*) AS secondary_sales,

        SUM(gross_price_sol)
            AS secondary_volume_sol,

        COUNT(DISTINCT buyer)
            AS unique_buyers,

        COUNT(DISTINCT mint)
            AS heroes_ever_sold,

        ROUND(
            SUM(
                COALESCE(
                    CAST(royalty_90_sol AS DOUBLE),
                    0.0
                )
            ),
            2
        ) AS guild_saga_royalties_sol

    FROM final_sales
),

/* ============================================================
   COMPACT MONTHLY VISUAL DATA

   This is equivalent to the working Market Visuals query.
   ============================================================ */

monthly_source AS (

    SELECT
        DATE_TRUNC(
            'month',
            FROM_UNIXTIME(
                CAST(block_time AS BIGINT)
            )
        ) AS month,

        COUNT(*) AS sales,

        ROUND(
            SUM(
                CAST(gross_price_sol AS DOUBLE)
            ),
            2
        ) AS volume_sol,

        ROUND(
            SUM(
                COALESCE(
                    CAST(royalty_90_sol AS DOUBLE),
                    0.0
                )
            ),
            2
        ) AS royalties_sol

    FROM final_sales

    GROUP BY 1
),

monthly_visuals AS (

    SELECT
        ROW_NUMBER() OVER (
            ORDER BY month
        ) AS viz_month_order,

        month,
        sales,
        volume_sol,
        royalties_sol

    FROM monthly_source
),

/* ============================================================
   FIRST RESALE

   The historical dataset already establishes the canonical
   distribution through the baseline.

   Any Hero that was "Never sold" there and now appears in
   final_sales has received its first post-baseline sale.

   All such future sales are >1y after mint.
   ============================================================ */

live_first_sales_from_never AS (

    SELECT
        COUNT(DISTINCT f.mint)
            AS moved_to_gt1y

    FROM final_sales AS f

    INNER JOIN
        dune.cjohnson.dataset_guild_saga_first_resale_timing AS r
        ON r.mint = f.mint

    WHERE
        r.first_resale_bucket = 'Never sold'
),

resale_visuals AS (

    SELECT
        v.resale_order,
        v.first_resale_bucket,

        CASE
            WHEN v.resale_order = 6
                THEN
                    v.baseline_heroes
                    + x.moved_to_gt1y

            WHEN v.resale_order = 7
                THEN
                    v.baseline_heroes
                    - x.moved_to_gt1y

            ELSE
                v.baseline_heroes

        END AS heroes

    FROM (
        VALUES
            (
                BIGINT '1',
                CAST('≤24h' AS VARCHAR),
                BIGINT '3866'
            ),
            (
                BIGINT '2',
                CAST('2–7d' AS VARCHAR),
                BIGINT '1236'
            ),
            (
                BIGINT '3',
                CAST('8–30d' AS VARCHAR),
                BIGINT '832'
            ),
            (
                BIGINT '4',
                CAST('31–90d' AS VARCHAR),
                BIGINT '1113'
            ),
            (
                BIGINT '5',
                CAST('91–365d' AS VARCHAR),
                BIGINT '948'
            ),
            (
                BIGINT '6',
                CAST('>1y' AS VARCHAR),
                BIGINT '410'
            ),
            (
                BIGINT '7',
                CAST('Never sold' AS VARCHAR),
                BIGINT '1495'
            )
    ) AS v (
        resale_order,
        first_resale_bucket,
        baseline_heroes
    )

    CROSS JOIN
        live_first_sales_from_never AS x
),

/* ============================================================
   PRESENTATION ROWS

   Roughly 55 rows total.

   First seven monthly rows also carry the seven resale buckets.
   ============================================================ */

presentation_output AS (

    SELECT
        CAST(NULL AS VARCHAR)
            AS signature,

        CAST(NULL AS VARCHAR)
            AS mint,

        CAST(NULL AS INTEGER)
            AS hero_number,

        CAST(NULL AS VARCHAR)
            AS hero_name,

        CAST(NULL AS BIGINT)
            AS block_time,

        CAST(NULL AS VARCHAR)
            AS utc,

        CAST(NULL AS VARCHAR)
            AS sale_date,

        CAST(NULL AS VARCHAR)
            AS sale_month,

        CAST(NULL AS INTEGER)
            AS sale_year,

        CAST(NULL AS VARCHAR)
            AS marketplace,

        CAST(NULL AS VARCHAR)
            AS marketplace_detail,

        CAST(NULL AS VARCHAR)
            AS marketplace_attribution_method,

        CAST(NULL AS VARCHAR)
            AS direct_program_families,

        CAST(NULL AS VARCHAR)
            AS buyer,

        CAST(NULL AS VARCHAR)
            AS seller,

        CAST(NULL AS DOUBLE)
            AS gross_price_sol,

        CAST(NULL AS VARCHAR)
            AS gross_price_method,

        CAST(NULL AS INTEGER)
            AS price_recovered_in_v3,

        CAST(NULL AS VARCHAR)
            AS sale_class,

        CAST(NULL AS INTEGER)
            AS legacy_sale,

        CAST(NULL AS DOUBLE)
            AS royalty_90_sol,

        CAST(NULL AS DOUBLE)
            AS royalty_10_sol,

        CAST(NULL AS DOUBLE)
            AS royalty_total_sol,

        CAST(NULL AS INTEGER)
            AS royalty_paid,

        CURRENT_DATE
            AS collector_checkpoint_date,

        /* ---------------- Market KPI cards ---------------- */

        CASE
            WHEN mv.viz_month_order = 1
            THEN m.secondary_sales
        END AS kpi_secondary_sales,

        CASE
            WHEN mv.viz_month_order = 1
            THEN m.secondary_volume_sol
        END AS kpi_secondary_volume_sol,

        CASE
            WHEN mv.viz_month_order = 1
            THEN m.unique_buyers
        END AS kpi_unique_buyers,

        CASE
            WHEN mv.viz_month_order = 1
            THEN m.heroes_ever_sold
        END AS kpi_heroes_ever_sold,

        /* ---------------- Monthly chart ---------------- */

        mv.month,
        mv.sales,
        mv.volume_sol,
        mv.royalties_sol,

        /* ---------------- Royalty cards ---------------- */

        CASE
            WHEN mv.viz_month_order = 1
            THEN m.guild_saga_royalties_sol
        END AS kpi_royalties_sol,

        CASE
            WHEN mv.viz_month_order = 1
            THEN ROUND(
                14850.0
                + m.guild_saga_royalties_sol,
                2
            )
        END AS kpi_mint_plus_royalties_sol,

        /* ---------------- First resale ---------------- */

        rv.resale_order,
        rv.first_resale_bucket,
        rv.heroes,

        BIGINT '1'
            AS output_section,

        mv.viz_month_order
            AS output_order

    FROM monthly_visuals AS mv

    CROSS JOIN metrics AS m

    LEFT JOIN resale_visuals AS rv
        ON rv.resale_order =
            mv.viz_month_order
),

/* ============================================================
   LIVE-SALE PERSISTENCE

   The 19,780 historical baseline DOES NOT need to be emitted.
   It is already permanently available in the baseline dataset
   and is rebuilt internally every run.

   Only new post-baseline sales must survive into
   previous.query.result().
   ============================================================ */

live_persistence_ranked AS (

    SELECT
        f.*,

        ROW_NUMBER() OVER (
            ORDER BY
                block_time DESC,
                signature,
                mint
        ) AS live_output_order

    FROM final_sales AS f

    WHERE
        sale_class = 'LIVE_DETECTED_SALE'
),

live_persistence_output AS (

        sale_month,

        sale_class,
        legacy_sale,

        royalty_90_sol,
        royalty_10_sol,
        royalty_total_sol,
        royalty_paid,

        CURRENT_DATE
            AS collector_checkpoint_date,

        CAST(NULL AS BIGINT)
            AS kpi_secondary_sales,

        CAST(NULL AS DOUBLE)
            AS kpi_secondary_volume_sol,

        CAST(NULL AS BIGINT)
            AS kpi_unique_buyers,

        CAST(NULL AS BIGINT)
            AS kpi_heroes_ever_sold,

        CAST(NULL AS TIMESTAMP)

        CAST(NULL AS BIGINT)
            AS heroes,

            AS output_section,

        live_output_order
            AS output_order

    FROM live_persistence_ranked
),

   FINAL COMPACT RESULT
   ============================================================ */

compact_output AS (

    SELECT *
    FROM presentation_output

    UNION ALL

    SELECT *
    FROM live_persistence_output
