SELECT
    CASE mint_phase
        WHEN 'PHASE_I'   THEN 'Phase I'
        WHEN 'PHASE_II'  THEN 'Phase II'
        WHEN 'PHASE_III' THEN 'Phase III'
    END AS "Mint Phase",

    COUNT(*) AS "Heroes Minted",

    COUNT(DISTINCT genesis_payer) AS "Unique Minters",

    MIN(
        CAST(from_iso8601_timestamp(genesis_utc) AS timestamp)
    ) AS "First Mint",

    MAX(
        CAST(from_iso8601_timestamp(genesis_utc) AS timestamp)
    ) AS "Last Mint"

FROM dune.cjohnson.dataset_guild_saga_assets_v2

WHERE genesis_type = 'public_candy_machine'

GROUP BY
    mint_phase

ORDER BY
    CASE mint_phase
        WHEN 'PHASE_I' THEN 1
        WHEN 'PHASE_II' THEN 2
        WHEN 'PHASE_III' THEN 3
    END;
