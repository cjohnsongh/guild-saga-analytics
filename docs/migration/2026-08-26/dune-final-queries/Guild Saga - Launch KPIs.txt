SELECT
    COUNT(*) AS public_mint_supply,

    COUNT(DISTINCT genesis_payer) AS unique_public_minters,

    CAST(1.5 AS DOUBLE) AS mint_price_sol,

    CAST(COUNT(*) * 1.5 AS DOUBLE) AS public_mint_proceeds_sol

FROM dune.cjohnson.dataset_guild_saga_assets_v2

WHERE genesis_type = 'public_candy_machine';
