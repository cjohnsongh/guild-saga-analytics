WITH wallet_mints AS (
    SELECT
        genesis_payer AS minter,
        COUNT(*) AS heroes_minted
    FROM dune.cjohnson.dataset_guild_saga_assets_v2
    WHERE genesis_type = 'public_candy_machine'
      AND genesis_payer IS NOT NULL
      AND genesis_payer <> ''
    GROUP BY 1
)

SELECT
    heroes_minted AS "Heroes Minted",
    COUNT(*) AS "Minters",
    SUM(heroes_minted) AS "Total Heroes"
FROM wallet_mints
GROUP BY 1
ORDER BY 1;
