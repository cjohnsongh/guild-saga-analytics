WITH prev AS (
    SELECT *
    FROM TABLE(
        previous.query.result(
            schema => DESCRIPTOR(
                row_type VARCHAR,

                mint VARCHAR,
                burned INTEGER,
                burn_utc VARCHAR,
                burn_signature VARCHAR,

                current_raw_owner VARCHAR,
                current_world_staked INTEGER,
                current_beneficial_owner VARCHAR,
                current_world_staking_wallet VARCHAR,

                latest_event_utc VARCHAR,
                latest_signature VARCHAR,

                quest_user_wallet VARCHAR,
                quest_staking_wallet VARCHAR,

                current_stake_deposit_utc VARCHAR,
                current_stake_deposit_signature VARCHAR,

                best_known_last_qualifying_quest_utc VARCHAR,
                best_known_last_qualifying_quest_signature VARCHAR,

                quest_history_source VARCHAR,
                deep_history_status VARCHAR,

                state_checkpoint_time TIMESTAMP(3) WITH TIME ZONE,

                kpi_active_supply BIGINT,
                kpi_beneficial_holders BIGINT,
                kpi_staked_heroes BIGINT,
                kpi_staked_supply_pct DOUBLE,
                kpi_quest_active_30d BIGINT,
                kpi_quest_limbo_1y BIGINT,

                viz_burn_month DATE,
                viz_monthly_burns BIGINT,
                viz_total_burns BIGINT,

                viz_rarity_order INTEGER,
                viz_rarity_level VARCHAR,
                viz_rarity_burns BIGINT,

                viz_quest_order INTEGER,
                viz_quest_bucket VARCHAR,
                viz_quest_heroes BIGINT,

                viz_holder_order INTEGER,
                viz_holder_tier VARCHAR,
                viz_holder_count BIGINT,
                viz_holder_supply BIGINT,
                viz_holder_supply_pct DOUBLE
            )
        )
    )
),

checkpoint AS (
    SELECT
        COALESCE(
            MAX(state_checkpoint_time),
            from_iso8601_timestamp(
                '2026-08-18T00:00:00Z'
            )
        ) - INTERVAL '2' DAY AS cutoff
    FROM prev
),

/* ============================================================
   COMPACT PERSISTED HERO STATE

   Only Heroes whose event-driven state differs from the
   immutable Aug. 18 baseline are persisted between runs.
   Presentation rows are ignored here.
   ============================================================ */

prev_state AS (
    SELECT
        mint,
        burned,
        burn_utc,
        burn_signature,
        current_raw_owner,
        current_world_staked,
        current_beneficial_owner,
        current_world_staking_wallet,
        latest_event_utc,
        latest_signature,
        quest_user_wallet,
        quest_staking_wallet,
        current_stake_deposit_utc,
        current_stake_deposit_signature,
        best_known_last_qualifying_quest_utc,
        best_known_last_qualifying_quest_signature,
        quest_history_source,
        deep_history_status,
        state_checkpoint_time
    FROM prev
    WHERE
        row_type = 'STATE_DELTA'
        AND mint IS NOT NULL
),

/* ============================================================
   HERO SEED

   Always start from the immutable 10,000-Hero baseline, then
   overlay only the compact persisted state deltas.
   ============================================================ */

seed AS (
    SELECT
        b.mint,
        b.hero_number,
        b.hero_name,
        b.genesis_type,
        b.genesis_utc,
        b.genesis_signature,
        b.genesis_payer,
        b.genesis_initial_owner,
        b.public_minter,
        b.mint_price_sol,
        b.mint_phase,

        CASE WHEN p.mint IS NOT NULL THEN p.burned
             ELSE b.burned END AS burned,
        CASE WHEN p.mint IS NOT NULL THEN p.burn_utc
             ELSE b.burn_utc END AS burn_utc,
        CASE WHEN p.mint IS NOT NULL THEN p.burn_signature
             ELSE b.burn_signature END AS burn_signature,

        CASE WHEN p.mint IS NOT NULL THEN p.current_raw_owner
             ELSE b.current_raw_owner END AS current_raw_owner,
        CASE WHEN p.mint IS NOT NULL THEN p.current_world_staked
             ELSE b.current_world_staked END AS current_world_staked,
        CASE WHEN p.mint IS NOT NULL THEN p.current_beneficial_owner
             ELSE b.current_beneficial_owner END AS current_beneficial_owner,
        CASE WHEN p.mint IS NOT NULL THEN p.current_world_staking_wallet
             ELSE b.current_world_staking_wallet END AS current_world_staking_wallet,

        CASE WHEN p.mint IS NOT NULL THEN p.latest_event_utc
             ELSE b.latest_event_utc END AS latest_event_utc,
        CASE WHEN p.mint IS NOT NULL THEN p.latest_signature
             ELSE b.latest_signature END AS latest_signature,

        CASE WHEN p.mint IS NOT NULL THEN p.quest_user_wallet
             ELSE b.quest_user_wallet END AS quest_user_wallet,
        CASE WHEN p.mint IS NOT NULL THEN p.quest_staking_wallet
             ELSE b.quest_staking_wallet END AS quest_staking_wallet,

        CASE WHEN p.mint IS NOT NULL THEN p.current_stake_deposit_utc
             ELSE b.current_stake_deposit_utc END AS current_stake_deposit_utc,
        CASE WHEN p.mint IS NOT NULL THEN p.current_stake_deposit_signature
             ELSE b.current_stake_deposit_signature END AS current_stake_deposit_signature,

        CASE WHEN p.mint IS NOT NULL THEN p.best_known_last_qualifying_quest_utc
             ELSE b.best_known_last_qualifying_quest_utc END
            AS best_known_last_qualifying_quest_utc,
        CASE WHEN p.mint IS NOT NULL THEN p.best_known_last_qualifying_quest_signature
             ELSE b.best_known_last_qualifying_quest_signature END
            AS best_known_last_qualifying_quest_signature,

        b.days_since_best_known_quest,
        b.quest_activity_status,

        CASE WHEN p.mint IS NOT NULL THEN p.quest_history_source
             ELSE b.quest_history_source END AS quest_history_source,

        b.no_qualifying_quest_current_stay,
        b.effective_quest_limbo_start_utc,
        b.effective_quest_limbo_start_year,

        CASE WHEN p.mint IS NOT NULL THEN p.deep_history_status
             ELSE b.deep_history_status END AS deep_history_status,

        p.state_checkpoint_time

    FROM dune.cjohnson.dataset_guild_saga_assets_v2 AS b

    LEFT JOIN prev_state AS p
        ON p.mint = b.mint
),

seed_parsed AS (
    SELECT
        s.*,

        COALESCE(
            TRY(
                from_iso8601_timestamp(
                    s.latest_event_utc
                )
            ),

            TRY_CAST(
                s.latest_event_utc
                AS TIMESTAMP WITH TIME ZONE
            ),

            TRY(
                with_timezone(
                    TRY_CAST(
                        s.latest_event_utc
                        AS TIMESTAMP
                    ),
                    'UTC'
                )
            )
        ) AS seed_latest_event_time_utc

    FROM seed AS s
),

/* ============================================================
   RECENT HERO TOKEN MOVEMENTS
   ============================================================ */

recent_raw_all AS (
    SELECT
        t.block_time,

        t.block_time AT TIME ZONE 'UTC'
            AS event_time_utc,

        t.block_date,
        t.block_slot,
        t.tx_index,
        t.outer_instruction_index,
        t.inner_instruction_index,

        t.tx_id,
        t.action,

        t.token_mint_address AS mint,
        t.from_owner,
        t.to_owner

    FROM tokens_solana.transfers AS t

    INNER JOIN
        dune.cjohnson.dataset_guild_saga_assets_v2 AS h
        ON h.mint = t.token_mint_address

    WHERE
        t.block_date >= CAST(
            (SELECT cutoff FROM checkpoint)
            AS DATE
        )

        AND (
            t.block_time AT TIME ZONE 'UTC'
        ) >= (
            SELECT cutoff
            FROM checkpoint
        )

        AND t.action IN (
            'transfer',
            'burn'
        )
),

latest_raw AS (
    SELECT
        block_time,
        event_time_utc,
        block_date,
        block_slot,
        tx_index,
        outer_instruction_index,
        inner_instruction_index,

        tx_id,
        action,
        mint,
        from_owner,
        to_owner

    FROM (
        SELECT
            r.*,

            ROW_NUMBER() OVER (
                PARTITION BY mint

                ORDER BY
                    block_time DESC,
                    block_slot DESC,
                    tx_index DESC,
                    COALESCE(
                        outer_instruction_index,
                        -1
                    ) DESC,
                    COALESCE(
                        inner_instruction_index,
                        -1
                    ) DESC
            ) AS rn

        FROM recent_raw_all AS r
    )

    WHERE rn = 1
),

/* ============================================================
   WORLD MODE
   ============================================================ */

world_calls AS (
    SELECT
        i.block_time,

        i.block_time AT TIME ZONE 'UTC'
            AS event_time_utc,

        i.block_date,
        i.tx_id,
        i.tx_signer AS user_wallet,

        CASE
            WHEN varbinary_substring(
                i.data,
                1,
                8
            ) = 0x5de284a68d093065
            THEN 'STAKE'

            WHEN varbinary_substring(
                i.data,
                1,
                8
            ) = 0x8eb5bf9552afd864
            THEN 'UNSTAKE'

            WHEN varbinary_substring(
                i.data,
                1,
                9
            ) = 0x042211a8aa29e88801
            THEN 'QUEST_RESTART'
        END AS world_action,

        CASE
            WHEN varbinary_substring(
                i.data,
                1,
                8
            ) IN (
                0x5de284a68d093065,
                0x8eb5bf9552afd864
            )
            THEN element_at(
                i.account_arguments,
                4
            )

            ELSE CAST(NULL AS VARCHAR)
        END AS staking_wallet

    FROM solana.instruction_calls AS i

    WHERE
        i.block_date >= CAST(
            (SELECT cutoff FROM checkpoint)
            AS DATE
        )

        AND (
            i.block_time AT TIME ZONE 'UTC'
        ) >= (
            SELECT cutoff
            FROM checkpoint
        )

        AND i.executing_account =
            '6AzuBKDsR88vinh399HV5v7fgB1eZyoYwQ3PmdYqFRZG'

        AND i.tx_success = TRUE

        AND (
            varbinary_substring(
                i.data,
                1,
                8
            ) IN (
                0x5de284a68d093065,
                0x8eb5bf9552afd864
            )

            OR

            varbinary_substring(
                i.data,
                1,
                9
            ) = 0x042211a8aa29e88801
        )
),

world_custody_candidates AS (
    SELECT
        w.event_time_utc,
        w.block_time,
        w.tx_id,

        w.world_action,
        w.user_wallet,
        w.staking_wallet,

        r.mint,
        r.from_owner,
        r.to_owner

    FROM world_calls AS w

    INNER JOIN recent_raw_all AS r
        ON r.tx_id = w.tx_id
        AND r.action = 'transfer'

    WHERE
        (
            w.world_action = 'STAKE'
            AND r.to_owner = w.staking_wallet
        )

        OR

        (
            w.world_action = 'UNSTAKE'
            AND r.from_owner = w.staking_wallet
            AND r.to_owner = w.user_wallet
        )
),

latest_world_custody AS (
    SELECT
        event_time_utc,
        block_time,
        tx_id,

        world_action,
        user_wallet,
        staking_wallet,

        mint,
        from_owner,
        to_owner

    FROM (
        SELECT
            w.*,

            ROW_NUMBER() OVER (
                PARTITION BY mint
                ORDER BY
                    event_time_utc DESC,
                    tx_id DESC
            ) AS rn

        FROM world_custody_candidates AS w
    )

    WHERE rn = 1
),

latest_quest AS (
    SELECT
        event_time_utc,
        block_time,
        tx_id,
        user_wallet

    FROM (
        SELECT
            w.event_time_utc,
            w.block_time,
            w.tx_id,
            w.user_wallet,

            ROW_NUMBER() OVER (
                PARTITION BY w.user_wallet
                ORDER BY
                    w.event_time_utc DESC,
                    w.tx_id DESC
            ) AS rn

        FROM world_calls AS w

        WHERE w.world_action = 'QUEST_RESTART'
    )

    WHERE rn = 1
),

/* ============================================================
   JOIN EVENTS TO CURRENT STATE
   ============================================================ */

movement_joined AS (
    SELECT
        s.*,

        r.event_time_utc AS raw_event_time_utc,
        r.tx_id AS raw_tx_id,
        r.action AS raw_action,
        r.from_owner AS raw_from_owner,
        r.to_owner AS raw_to_owner,

        w.event_time_utc AS world_event_time_utc,
        w.tx_id AS world_tx_id,
        w.world_action AS world_action,
        w.user_wallet AS world_user_wallet,
        w.staking_wallet AS world_staking_wallet

    FROM seed_parsed AS s

    LEFT JOIN latest_raw AS r
        ON r.mint = s.mint

    LEFT JOIN latest_world_custody AS w
        ON w.mint = s.mint
),

movement_flags AS (
    SELECT
        m.*,

        CASE
            WHEN raw_tx_id IS NULL
            THEN 0

            WHEN seed_latest_event_time_utc IS NULL
            THEN 1

            WHEN raw_event_time_utc
                >= seed_latest_event_time_utc
            THEN 1

            ELSE 0
        END AS raw_relevant,

        CASE
            WHEN raw_tx_id IS NULL
            THEN 0

            WHEN seed_latest_event_time_utc IS NOT NULL
             AND raw_event_time_utc
                    < seed_latest_event_time_utc
            THEN 0

            WHEN raw_tx_id
                IS DISTINCT FROM latest_signature
            THEN 1

            ELSE 0
        END AS raw_new

    FROM movement_joined AS m
),

custody_flags AS (
    SELECT
        m.*,

        CASE
            WHEN raw_relevant = 1
             AND world_tx_id IS NOT NULL
             AND world_tx_id = raw_tx_id
            THEN 1
            ELSE 0
        END AS custody_matches_raw

    FROM movement_flags AS m
),

transition_flags AS (
    SELECT
        m.*,

        CASE
            WHEN custody_matches_raw = 1
             AND world_action = 'STAKE'

             AND (
                    raw_new = 1

                 OR current_world_staked <> 1

                 OR current_world_staking_wallet
                    IS DISTINCT FROM world_staking_wallet

                 OR current_beneficial_owner
                    IS DISTINCT FROM world_user_wallet
             )
            THEN 1
            ELSE 0
        END AS stake_transition,

        CASE
            WHEN custody_matches_raw = 1
             AND world_action = 'UNSTAKE'

             AND (
                    raw_new = 1

                 OR current_world_staked = 1

                 OR current_world_staking_wallet
                    IS NOT NULL
             )
            THEN 1
            ELSE 0
        END AS unstake_transition,

        CASE
            WHEN raw_new = 1
             AND custody_matches_raw = 0
            THEN 1
            ELSE 0
        END AS ordinary_movement_transition,

        CASE
            WHEN raw_new = 1
             AND raw_action = 'burn'
            THEN 1
            ELSE 0
        END AS burn_transition

    FROM custody_flags AS m
),

/* ============================================================
   APPLY RAW + WORLD MODE STATE
   ============================================================ */

movement_applied AS (
    SELECT
        mint,
        hero_number,
        hero_name,
        genesis_type,
        genesis_utc,
        genesis_signature,
        genesis_payer,
        genesis_initial_owner,
        public_minter,
        mint_price_sol,
        mint_phase,

        CASE
            WHEN burned = 1
            THEN CAST(1 AS INTEGER)

            WHEN burn_transition = 1
            THEN CAST(1 AS INTEGER)

            ELSE burned
        END AS burned_after,

        CASE
            WHEN burn_transition = 1
            THEN to_iso8601(raw_event_time_utc)
            ELSE burn_utc
        END AS burn_utc_after,

        CASE
            WHEN burn_transition = 1
            THEN raw_tx_id
            ELSE burn_signature
        END AS burn_signature_after,

        CASE
            WHEN raw_new = 1
            THEN raw_to_owner
            ELSE current_raw_owner
        END AS current_raw_owner_after,

        CASE
            WHEN burn_transition = 1
            THEN CAST(0 AS INTEGER)

            WHEN stake_transition = 1
            THEN CAST(1 AS INTEGER)

            WHEN unstake_transition = 1
            THEN CAST(0 AS INTEGER)

            WHEN ordinary_movement_transition = 1
            THEN CAST(0 AS INTEGER)

            ELSE current_world_staked
        END AS current_world_staked_after,

        CASE
            WHEN burn_transition = 1
            THEN CAST(NULL AS VARCHAR)

            WHEN stake_transition = 1
            THEN world_user_wallet

            WHEN unstake_transition = 1
            THEN raw_to_owner

            WHEN ordinary_movement_transition = 1
            THEN raw_to_owner

            ELSE current_beneficial_owner
        END AS current_beneficial_owner_after,

        CASE
            WHEN burn_transition = 1
            THEN CAST(NULL AS VARCHAR)

            WHEN stake_transition = 1
            THEN world_staking_wallet

            WHEN unstake_transition = 1
            THEN CAST(NULL AS VARCHAR)

            WHEN ordinary_movement_transition = 1
            THEN CAST(NULL AS VARCHAR)

            ELSE current_world_staking_wallet
        END AS current_world_staking_wallet_after,

        CASE
            WHEN raw_new = 1
            THEN to_iso8601(raw_event_time_utc)
            ELSE latest_event_utc
        END AS latest_event_utc_after,

        CASE
            WHEN raw_new = 1
            THEN raw_tx_id
            ELSE latest_signature
        END AS latest_signature_after,

        CASE
            WHEN burn_transition = 1
            THEN CAST(NULL AS VARCHAR)

            WHEN stake_transition = 1
            THEN world_user_wallet

            WHEN unstake_transition = 1
            THEN CAST(NULL AS VARCHAR)

            WHEN ordinary_movement_transition = 1
            THEN CAST(NULL AS VARCHAR)

            ELSE quest_user_wallet
        END AS quest_user_wallet_after,

        CASE
            WHEN burn_transition = 1
            THEN CAST(NULL AS VARCHAR)

            WHEN stake_transition = 1
            THEN world_staking_wallet

            WHEN unstake_transition = 1
            THEN CAST(NULL AS VARCHAR)

            WHEN ordinary_movement_transition = 1
            THEN CAST(NULL AS VARCHAR)

            ELSE quest_staking_wallet
        END AS quest_staking_wallet_after,

        CASE
            WHEN burn_transition = 1
            THEN CAST(NULL AS VARCHAR)

            WHEN stake_transition = 1
            THEN to_iso8601(world_event_time_utc)

            WHEN unstake_transition = 1
            THEN CAST(NULL AS VARCHAR)

            WHEN ordinary_movement_transition = 1
            THEN CAST(NULL AS VARCHAR)

            ELSE current_stake_deposit_utc
        END AS current_stake_deposit_utc_after,

        CASE
            WHEN burn_transition = 1
            THEN CAST(NULL AS VARCHAR)

            WHEN stake_transition = 1
            THEN world_tx_id

            WHEN unstake_transition = 1
            THEN CAST(NULL AS VARCHAR)

            WHEN ordinary_movement_transition = 1
            THEN CAST(NULL AS VARCHAR)

            ELSE current_stake_deposit_signature
        END AS current_stake_deposit_signature_after,

        CASE
            WHEN burn_transition = 1
              OR stake_transition = 1
              OR unstake_transition = 1
              OR ordinary_movement_transition = 1
            THEN CAST(NULL AS VARCHAR)

            ELSE best_known_last_qualifying_quest_utc
        END AS prior_qualifying_quest_utc_after,

        CASE
            WHEN burn_transition = 1
              OR stake_transition = 1
              OR unstake_transition = 1
              OR ordinary_movement_transition = 1
            THEN CAST(NULL AS VARCHAR)

            ELSE best_known_last_qualifying_quest_signature
        END AS prior_qualifying_quest_signature_after,

        CASE
            WHEN burn_transition = 1
              OR unstake_transition = 1
              OR ordinary_movement_transition = 1
            THEN CAST(NULL AS VARCHAR)

            WHEN stake_transition = 1
            THEN 'DUNE_LIVE'

            ELSE quest_history_source
        END AS quest_history_source_after,

        CASE
            WHEN burn_transition = 1
              OR unstake_transition = 1
              OR ordinary_movement_transition = 1
            THEN CAST(NULL AS VARCHAR)

            WHEN stake_transition = 1
            THEN 'LIVE_POST_BASELINE'

            ELSE deep_history_status
        END AS deep_history_status_after

    FROM transition_flags
),

movement_parsed AS (
    SELECT
        m.*,

        COALESCE(
            TRY(
                from_iso8601_timestamp(
                    current_stake_deposit_utc_after
                )
            ),

            TRY_CAST(
                current_stake_deposit_utc_after
                AS TIMESTAMP WITH TIME ZONE
            ),

            TRY(
                with_timezone(
                    TRY_CAST(
                        current_stake_deposit_utc_after
                        AS TIMESTAMP
                    ),
                    'UTC'
                )
            )
        ) AS current_stake_time_utc,

        COALESCE(
            TRY(
                from_iso8601_timestamp(
                    prior_qualifying_quest_utc_after
                )
            ),

            TRY_CAST(
                prior_qualifying_quest_utc_after
                AS TIMESTAMP WITH TIME ZONE
            ),

            TRY(
                with_timezone(
                    TRY_CAST(
                        prior_qualifying_quest_utc_after
                        AS TIMESTAMP
                    ),
                    'UTC'
                )
            )
        ) AS prior_quest_time_utc

    FROM movement_applied AS m
),

quest_joined AS (
    SELECT
        m.*,

        q.event_time_utc AS new_quest_time_utc,
        q.tx_id AS new_quest_tx

    FROM movement_parsed AS m

    LEFT JOIN latest_quest AS q
        ON q.user_wallet = m.quest_user_wallet_after
),

quest_evaluated AS (
    SELECT
        q.*,

        CASE
            WHEN current_world_staked_after = 1
             AND new_quest_time_utc IS NOT NULL
             AND current_stake_time_utc IS NOT NULL
             AND new_quest_time_utc >= current_stake_time_utc

             AND (
                    prior_quest_time_utc IS NULL
                 OR new_quest_time_utc > prior_quest_time_utc
             )
            THEN 1
            ELSE 0
        END AS quest_advances

    FROM quest_joined AS q
),

quest_effective AS (
    SELECT
        q.*,

        CASE
            WHEN quest_advances = 1
            THEN new_quest_time_utc
            ELSE prior_quest_time_utc
        END AS effective_quest_time_utc,

        CASE
            WHEN quest_advances = 1
            THEN new_quest_tx
            ELSE prior_qualifying_quest_signature_after
        END AS effective_quest_signature

    FROM quest_evaluated AS q
),

aged AS (
    SELECT
        q.*,

        CASE
            WHEN current_world_staked_after = 1
             AND effective_quest_time_utc IS NOT NULL
            THEN
                CAST(
                    date_diff(
                        'second',
                        effective_quest_time_utc,
                        NOW()
                    )
                    AS DOUBLE
                ) / 86400.0

            ELSE CAST(NULL AS DOUBLE)
        END AS effective_days_since_quest

    FROM quest_effective AS q
),

/* ============================================================
   HEADLINE KPIs
   ============================================================ */

kpis AS (
    SELECT
        COUNT_IF(
            burned_after = 0
        ) AS kpi_active_supply,

        COUNT(
            DISTINCT CASE
                WHEN burned_after = 0
                THEN current_beneficial_owner_after
            END
        ) AS kpi_beneficial_holders,

        COUNT_IF(
            current_world_staked_after = 1
        ) AS kpi_staked_heroes,

        CAST(
            ROUND(
                100.0
                * COUNT_IF(
                    current_world_staked_after = 1
                )
                / NULLIF(
                    COUNT_IF(
                        burned_after = 0
                    ),
                    0
                ),
                1
            )
            AS DOUBLE
        ) AS kpi_staked_supply_pct,

        COUNT_IF(
            current_world_staked_after = 1
            AND effective_quest_time_utc IS NOT NULL
            AND effective_days_since_quest <= 30
        ) AS kpi_quest_active_30d,

        COUNT_IF(
            current_world_staked_after = 1
            AND (
                effective_quest_time_utc IS NULL
                OR effective_days_since_quest > 365
            )
        ) AS kpi_quest_limbo_1y

    FROM aged
),

/* ============================================================
   BURN HISTORY VISUALIZATION DATA
   ============================================================ */

burn_events AS (
    SELECT
        COALESCE(
            TRY(
                from_iso8601_timestamp(
                    burn_utc_after
                )
            ),

            TRY_CAST(
                burn_utc_after
                AS TIMESTAMP WITH TIME ZONE
            ),

            TRY(
                with_timezone(
                    TRY_CAST(
                        burn_utc_after
                        AS TIMESTAMP
                    ),
                    'UTC'
                )
            )
        ) AS burn_time

    FROM aged

    WHERE
        burned_after = 1
        AND burn_utc_after IS NOT NULL
),

burn_monthly AS (
    SELECT
        CAST(
            DATE_TRUNC(
                'month',
                burn_time
            )
            AS DATE
        ) AS burn_month,

        COUNT(*) AS monthly_burns

    FROM burn_events

    WHERE burn_time IS NOT NULL

    GROUP BY 1
),

burn_month_bounds AS (
    SELECT
        MIN(burn_month) AS first_month,

        CAST(
            DATE_TRUNC(
                'month',
                CURRENT_DATE
            )
            AS DATE
        ) AS last_month

    FROM burn_monthly
),

burn_calendar AS (
    SELECT
        month AS burn_month

    FROM burn_month_bounds

    CROSS JOIN UNNEST(
        SEQUENCE(
            first_month,
            last_month,
            INTERVAL '1' MONTH
        )
    ) AS t(month)
),

burn_filled AS (
    SELECT
        c.burn_month,

        COALESCE(
            b.monthly_burns,
            CAST(0 AS BIGINT)
        ) AS monthly_burns

    FROM burn_calendar AS c

    LEFT JOIN burn_monthly AS b
        ON b.burn_month = c.burn_month
),

burn_summary AS (
    SELECT
        ROW_NUMBER() OVER (
            ORDER BY burn_month
        ) AS viz_row,

        burn_month AS viz_burn_month,

        monthly_burns AS viz_monthly_burns,

        SUM(monthly_burns) OVER (
            ORDER BY burn_month
            ROWS BETWEEN
                UNBOUNDED PRECEDING
                AND CURRENT ROW
        ) AS viz_total_burns

    FROM burn_filled
),

/* ============================================================
   BURNED HEROES BY RARITY

   Rarity is immutable/static and comes from the one-time
   10,000-mint upload. Burn state comes from the live Hero state.
   ============================================================ */

rarity_bucket_defs AS (
    SELECT *
    FROM (
        VALUES
            (1, 'Bronze'),
            (2, 'Silver'),
            (3, 'Gold'),
            (4, 'Arcane'),
            (5, 'Elven')
    ) AS t(
        rarity_order,
        rarity_level
    )
),

rarity_burn_counts AS (
    SELECT
        r.rarity_level,
        COUNT(*) AS burned_heroes

    FROM dune.cjohnson.dataset_guild_saga_hero_rarity_v1 AS r

    INNER JOIN aged AS a
        ON a.mint = r.mint

    WHERE a.burned_after = 1

    GROUP BY 1
),

rarity_summary AS (
    SELECT
        CAST(d.rarity_order AS INTEGER)
            AS viz_rarity_order,

        d.rarity_level
            AS viz_rarity_level,

        COALESCE(
            c.burned_heroes,
            CAST(0 AS BIGINT)
        ) AS viz_rarity_burns

    FROM rarity_bucket_defs AS d

    LEFT JOIN rarity_burn_counts AS c
        ON c.rarity_level = d.rarity_level
),

/* ============================================================
   QUEST ACTIVITY VISUALIZATION DATA
   ============================================================ */

quest_bucket_defs AS (
    SELECT *
    FROM (
        VALUES
            (1, 'Active 0–7d'),
            (2, 'Idle 8–30d'),
            (3, 'Idle 31–90d'),
            (4, 'Idle 91–180d'),
            (5, 'Idle 181–365d'),
            (6, 'Idle 1+ year'),
            (7, 'Never quested')
    ) AS t(
        bucket_order,
        bucket_name
    )
),

quest_bucketed AS (
    SELECT
        CASE
            WHEN effective_quest_time_utc IS NULL
            THEN 7

            WHEN effective_days_since_quest <= 7
            THEN 1

            WHEN effective_days_since_quest <= 30
            THEN 2

            WHEN effective_days_since_quest <= 90
            THEN 3

            WHEN effective_days_since_quest <= 180
            THEN 4

            WHEN effective_days_since_quest <= 365
            THEN 5

            ELSE 6
        END AS bucket_order

    FROM aged

    WHERE current_world_staked_after = 1
),

quest_counts AS (
    SELECT
        bucket_order,
        COUNT(*) AS heroes

    FROM quest_bucketed

    GROUP BY 1
),

quest_summary AS (
    SELECT
        CAST(
            d.bucket_order
            AS INTEGER
        ) AS viz_quest_order,

        d.bucket_name
            AS viz_quest_bucket,

        COALESCE(
            c.heroes,
            CAST(0 AS BIGINT)
        ) AS viz_quest_heroes

    FROM quest_bucket_defs AS d

    LEFT JOIN quest_counts AS c
        ON c.bucket_order = d.bucket_order
),

/* ============================================================
   HOLDER DISTRIBUTION VISUALIZATION DATA
   ============================================================ */

holder_balances AS (
    SELECT
        current_beneficial_owner_after
            AS holder,

        COUNT(*) AS heroes_held

    FROM aged

    WHERE
        burned_after = 0

        AND current_beneficial_owner_after
            IS NOT NULL

    GROUP BY 1
),

holder_bucketed AS (
    SELECT
        holder,
        heroes_held,

        CASE
            WHEN heroes_held <= 4
            THEN 1

            WHEN heroes_held <= 14
            THEN 2

            WHEN heroes_held <= 29
            THEN 3

            WHEN heroes_held <= 49
            THEN 4

            WHEN heroes_held <= 99
            THEN 5

            ELSE 6
        END AS bucket_order

    FROM holder_balances
),

holder_bucket_defs AS (
    SELECT *
    FROM (
        VALUES
            (1, '1-4'),
            (2, '5-14'),
            (3, '15-29'),
            (4, '30-49'),
            (5, '50-99'),
            (6, '100+')
    ) AS t(
        bucket_order,
        bucket_name
    )
),

holder_counts AS (
    SELECT
        bucket_order,

        COUNT(*) AS holder_count,

        SUM(heroes_held) AS holder_supply

    FROM holder_bucketed

    GROUP BY 1
),

holder_summary AS (
    SELECT
        CAST(
            d.bucket_order
            AS INTEGER
        ) AS viz_holder_order,

        d.bucket_name
            AS viz_holder_tier,

        COALESCE(
            c.holder_count,
            CAST(0 AS BIGINT)
        ) AS viz_holder_count,

        COALESCE(
            c.holder_supply,
            CAST(0 AS BIGINT)
        ) AS viz_holder_supply,

        CAST(
            ROUND(
                100.0
                * COALESCE(
                    c.holder_supply,
                    CAST(0 AS BIGINT)
                )
                / NULLIF(
                    k.kpi_active_supply,
                    0
                ),
                1
            )
            AS DOUBLE
        ) AS viz_holder_supply_pct

    FROM holder_bucket_defs AS d

    LEFT JOIN holder_counts AS c
        ON c.bucket_order = d.bucket_order

    CROSS JOIN kpis AS k
),

/* ============================================================
   NORMALIZED CURRENT HERO STATE

   This is the same live state V7 produced, but it exists only
   inside the query. We no longer emit 10,000 Hero rows.
   ============================================================ */

current_state AS (
    SELECT
        a.mint,

        a.burned_after AS burned,
        a.burn_utc_after AS burn_utc,
        a.burn_signature_after AS burn_signature,

        a.current_raw_owner_after
            AS current_raw_owner,

        a.current_world_staked_after
            AS current_world_staked,

        a.current_beneficial_owner_after
            AS current_beneficial_owner,

        a.current_world_staking_wallet_after
            AS current_world_staking_wallet,

        a.latest_event_utc_after
            AS latest_event_utc,

        a.latest_signature_after
            AS latest_signature,

        a.quest_user_wallet_after
            AS quest_user_wallet,

        a.quest_staking_wallet_after
            AS quest_staking_wallet,

        a.current_stake_deposit_utc_after
            AS current_stake_deposit_utc,

        a.current_stake_deposit_signature_after
            AS current_stake_deposit_signature,

        CASE
            WHEN a.current_world_staked_after <> 1
            THEN CAST(NULL AS VARCHAR)

            WHEN a.quest_advances = 1
            THEN to_iso8601(
                a.new_quest_time_utc
            )

            ELSE a.prior_qualifying_quest_utc_after
        END AS best_known_last_qualifying_quest_utc,

        CASE
            WHEN a.current_world_staked_after <> 1
            THEN CAST(NULL AS VARCHAR)

            ELSE a.effective_quest_signature
        END AS best_known_last_qualifying_quest_signature,

        CASE
            WHEN a.current_world_staked_after <> 1
            THEN CAST(NULL AS VARCHAR)

            WHEN a.quest_advances = 1
            THEN 'DUNE_LIVE'

            ELSE a.quest_history_source_after
        END AS quest_history_source,

        CASE
            WHEN a.current_world_staked_after <> 1
            THEN CAST(NULL AS VARCHAR)

            WHEN a.quest_advances = 1
            THEN 'LIVE_POST_BASELINE'

            ELSE a.deep_history_status_after
        END AS deep_history_status

    FROM aged AS a
),

/* ============================================================
   PERSIST ONLY STATE THAT DIFFERS FROM THE AUG. 18 BASELINE

   Time-derived fields such as days-since-quest and activity
   bucket are deliberately NOT persisted. They are recalculated
   from timestamps every run, so they continue aging naturally.
   ============================================================ */

state_deltas AS (
    SELECT
        c.*

    FROM current_state AS c

    INNER JOIN dune.cjohnson.dataset_guild_saga_assets_v2 AS b
        ON b.mint = c.mint

    WHERE
           c.burned
                IS DISTINCT FROM b.burned
        OR c.burn_utc
                IS DISTINCT FROM b.burn_utc
        OR c.burn_signature
                IS DISTINCT FROM b.burn_signature
        OR c.current_raw_owner
                IS DISTINCT FROM b.current_raw_owner
        OR c.current_world_staked
                IS DISTINCT FROM b.current_world_staked
        OR c.current_beneficial_owner
                IS DISTINCT FROM b.current_beneficial_owner
        OR c.current_world_staking_wallet
                IS DISTINCT FROM b.current_world_staking_wallet
        OR c.latest_event_utc
                IS DISTINCT FROM b.latest_event_utc
        OR c.latest_signature
                IS DISTINCT FROM b.latest_signature
        OR c.quest_user_wallet
                IS DISTINCT FROM b.quest_user_wallet
        OR c.quest_staking_wallet
                IS DISTINCT FROM b.quest_staking_wallet
        OR c.current_stake_deposit_utc
                IS DISTINCT FROM b.current_stake_deposit_utc
        OR c.current_stake_deposit_signature
                IS DISTINCT FROM b.current_stake_deposit_signature
        OR c.best_known_last_qualifying_quest_utc
                IS DISTINCT FROM b.best_known_last_qualifying_quest_utc
        OR c.best_known_last_qualifying_quest_signature
                IS DISTINCT FROM b.best_known_last_qualifying_quest_signature
        OR c.quest_history_source
                IS DISTINCT FROM b.quest_history_source
        OR c.deep_history_status
                IS DISTINCT FROM b.deep_history_status
),

/* ============================================================
   COMPACT PRESENTATION ROWS

   Burn history provides the row spine. Quest and holder
   summaries are packed into the first 7 / 6 rows respectively.
   This is the same compact-output pattern that fixed the Market
   Visuals scheduling problem.
   ============================================================ */

presentation_output AS (
    SELECT
        CAST('PRESENTATION' AS VARCHAR)
            AS row_type,

        CAST(NULL AS VARCHAR)
            AS mint,

        CAST(NULL AS INTEGER)
            AS burned,
        CAST(NULL AS VARCHAR)
            AS burn_utc,
        CAST(NULL AS VARCHAR)
            AS burn_signature,

        CAST(NULL AS VARCHAR)
            AS current_raw_owner,
        CAST(NULL AS INTEGER)
            AS current_world_staked,
        CAST(NULL AS VARCHAR)
            AS current_beneficial_owner,
        CAST(NULL AS VARCHAR)
            AS current_world_staking_wallet,

        CAST(NULL AS VARCHAR)
            AS latest_event_utc,
        CAST(NULL AS VARCHAR)
            AS latest_signature,

        CAST(NULL AS VARCHAR)
            AS quest_user_wallet,
        CAST(NULL AS VARCHAR)
            AS quest_staking_wallet,

        CAST(NULL AS VARCHAR)
            AS current_stake_deposit_utc,
        CAST(NULL AS VARCHAR)
            AS current_stake_deposit_signature,

        CAST(NULL AS VARCHAR)
            AS best_known_last_qualifying_quest_utc,
        CAST(NULL AS VARCHAR)
            AS best_known_last_qualifying_quest_signature,

        CAST(NULL AS VARCHAR)
            AS quest_history_source,
        CAST(NULL AS VARCHAR)
            AS deep_history_status,

        CAST(
            NOW()
            AS TIMESTAMP(3) WITH TIME ZONE
        ) AS state_checkpoint_time,

        CASE WHEN b.viz_row = 1
             THEN k.kpi_active_supply END
            AS kpi_active_supply,

        CASE WHEN b.viz_row = 1
             THEN k.kpi_beneficial_holders END
            AS kpi_beneficial_holders,

        CASE WHEN b.viz_row = 1
             THEN k.kpi_staked_heroes END
            AS kpi_staked_heroes,

        CASE WHEN b.viz_row = 1
             THEN k.kpi_staked_supply_pct END
            AS kpi_staked_supply_pct,

        CASE WHEN b.viz_row = 1
             THEN k.kpi_quest_active_30d END
            AS kpi_quest_active_30d,

        CASE WHEN b.viz_row = 1
             THEN k.kpi_quest_limbo_1y END
            AS kpi_quest_limbo_1y,

        b.viz_burn_month,
        b.viz_monthly_burns,
        b.viz_total_burns,

        r.viz_rarity_order,
        r.viz_rarity_level,
        r.viz_rarity_burns,

        q.viz_quest_order,
        q.viz_quest_bucket,
        q.viz_quest_heroes,

        h.viz_holder_order,
        h.viz_holder_tier,
        h.viz_holder_count,
        h.viz_holder_supply,
        h.viz_holder_supply_pct,

        CAST(1 AS INTEGER)
            AS output_section,

        CAST(b.viz_row AS BIGINT)
            AS output_order

    FROM burn_summary AS b

    CROSS JOIN kpis AS k

    LEFT JOIN rarity_summary AS r
        ON r.viz_rarity_order = b.viz_row

    LEFT JOIN quest_summary AS q
        ON q.viz_quest_order = b.viz_row

    LEFT JOIN holder_summary AS h
        ON h.viz_holder_order = b.viz_row
),

state_delta_numbered AS (
    SELECT
        d.*,

        ROW_NUMBER() OVER (
            ORDER BY d.mint
        ) AS delta_order

    FROM state_deltas AS d
),

state_delta_output AS (
    SELECT
        CAST('STATE_DELTA' AS VARCHAR)
            AS row_type,

        d.mint,

        d.burned,
        d.burn_utc,
        d.burn_signature,

        d.current_raw_owner,
        d.current_world_staked,
        d.current_beneficial_owner,
        d.current_world_staking_wallet,

        d.latest_event_utc,
        d.latest_signature,

        d.quest_user_wallet,
        d.quest_staking_wallet,

        d.current_stake_deposit_utc,
        d.current_stake_deposit_signature,

        d.best_known_last_qualifying_quest_utc,
        d.best_known_last_qualifying_quest_signature,

        d.quest_history_source,
        d.deep_history_status,

        CAST(
            NOW()
            AS TIMESTAMP(3) WITH TIME ZONE
        ) AS state_checkpoint_time,

        CAST(NULL AS BIGINT)
            AS kpi_active_supply,
        CAST(NULL AS BIGINT)
            AS kpi_beneficial_holders,
        CAST(NULL AS BIGINT)
            AS kpi_staked_heroes,
        CAST(NULL AS DOUBLE)
            AS kpi_staked_supply_pct,
        CAST(NULL AS BIGINT)
            AS kpi_quest_active_30d,
        CAST(NULL AS BIGINT)
            AS kpi_quest_limbo_1y,

        CAST(NULL AS DATE)
            AS viz_burn_month,
        CAST(NULL AS BIGINT)
            AS viz_monthly_burns,
        CAST(NULL AS BIGINT)
            AS viz_total_burns,

        CAST(NULL AS INTEGER)
            AS viz_rarity_order,
        CAST(NULL AS VARCHAR)
            AS viz_rarity_level,
        CAST(NULL AS BIGINT)
            AS viz_rarity_burns,

        CAST(NULL AS INTEGER)
            AS viz_quest_order,
        CAST(NULL AS VARCHAR)
            AS viz_quest_bucket,
        CAST(NULL AS BIGINT)
            AS viz_quest_heroes,

        CAST(NULL AS INTEGER)
            AS viz_holder_order,
        CAST(NULL AS VARCHAR)
            AS viz_holder_tier,
        CAST(NULL AS BIGINT)
            AS viz_holder_count,
        CAST(NULL AS BIGINT)
            AS viz_holder_supply,
        CAST(NULL AS DOUBLE)
            AS viz_holder_supply_pct,

        CAST(2 AS INTEGER)
            AS output_section,

        CAST(d.delta_order AS BIGINT)
            AS output_order

    FROM state_delta_numbered AS d
),

compact_output AS (
    SELECT *
    FROM presentation_output

    UNION ALL

    SELECT *
    FROM state_delta_output
)

SELECT
    row_type,

    mint,
    burned,
    burn_utc,
    burn_signature,

    current_raw_owner,
    current_world_staked,
    current_beneficial_owner,
    current_world_staking_wallet,

    latest_event_utc,
    latest_signature,

    quest_user_wallet,
    quest_staking_wallet,

    current_stake_deposit_utc,
    current_stake_deposit_signature,

    best_known_last_qualifying_quest_utc,
    best_known_last_qualifying_quest_signature,

    quest_history_source,
    deep_history_status,

    state_checkpoint_time,

    kpi_active_supply,
    kpi_beneficial_holders,
    kpi_staked_heroes,
    kpi_staked_supply_pct,
    kpi_quest_active_30d,
    kpi_quest_limbo_1y,

    viz_burn_month,
    viz_monthly_burns,
    viz_total_burns,

    viz_rarity_order,
    viz_rarity_level,
    viz_rarity_burns,

    viz_quest_order,
    viz_quest_bucket,
    viz_quest_heroes,

    viz_holder_order,
    viz_holder_tier,
    viz_holder_count,
    viz_holder_supply,
    viz_holder_supply_pct

FROM compact_output

ORDER BY
    output_section,
    output_order;
